\begin{tikzpicture}[
  >={Latex[length=3mm,width=2.2mm]},
  every node/.style={font=\small},
  posc/.style={circle,draw=black!60,fill=posred,inner sep=0pt,minimum size=5mm}
]

% --- (a) Carica al centro ---
\begin{scope}
  % Sfondo azzurro con trattini (campo B uniforme)
  \fill[surfblue!40] (-2.5,-3.0) rectangle (2.5,3.0);
  % Trattini orizzontali (B entrante nel foglio)
  \foreach \x in {-2.0,-1.2,...,2.0}{
    \foreach \y in {-2.5,-1.8,...,2.5}{
      \draw[thin,black] ({\x-0.15},\y) -- ({\x+0.15},\y);
    }
  }

  % Carica + al centro
  \node[posc] at (0,0) {\tiny$+$};

  \node[below] at (0,-3.2) {(a)};
\end{scope}

% --- (b) Carica spostata di z ---
\begin{scope}[yshift=-8cm]
  \fill[surfblue!40] (-2.5,-3.0) rectangle (2.5,3.0);
  \foreach \x in {-2.0,-1.2,...,2.0}{
    \foreach \y in {-2.5,-1.8,...,2.5}{
      \draw[thin,black] ({\x-0.15},\y) -- ({\x+0.15},\y);
    }
  }

  % Carica + spostata in alto
  \node[posc] at (0,0.6) {\tiny$+$};

  % Spostamento z
  \draw[<->,thin] (0.35,0) -- (0.35,0.6);
  \node[right] at (0.35,0.3) {$z$};
  \draw[thin] (-0.1,0) -- (0.5,0);

  \node[below] at (0,-3.2) {(b)};
\end{scope}

\end{tikzpicture}
