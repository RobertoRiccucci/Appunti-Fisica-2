\begin{tikzpicture}[
  >={Latex[length=3mm,width=2.2mm]},
  every node/.style={font=\small},
  bvec/.style={->,line width=1.2pt,bblue}
]

% Sfondo esterno (beige chiaro)
\fill[ytangent!50] (0,0) circle (3.0);

% Regione interna E (più scura, beige)
\fill[ytangent] (0,0) circle (1.8);

% Etichetta E al centro
\node[ered,font=\large] at (0,0) {$\mathbf{E}$};

% Punti E (uscenti dal foglio, rossi)
\foreach \x in {-1.0,0,1.0}{
  \foreach \y in {-1.0,0,1.0}{
    \fill[ered] (\x,\y) circle (2pt);
  }
}
\foreach \x in {-0.5,0.5}{
  \foreach \y in {-0.5,0.5}{
    \fill[ered] (\x,\y) circle (2pt);
  }
}
\foreach \a in {0,45,...,315}{
  \fill[ered] ({2.3*cos(\a)},{2.3*sin(\a)}) circle (2pt);
}

% Cerchio
\draw[thick,black] (0,0) circle (1.8);

% Vettori B tangenti al cerchio (blu)
\draw[bvec] (0,1.8)   -- ++(-0.8,0)  node[above left] {$\mathbf{B}$};
\draw[bvec] (-1.8,0)   -- ++(0,-0.8)  node[below left] {$\mathbf{B}$};
\draw[bvec] (1.8,0)    -- ++(0,0.8) node[above right] {$\mathbf{B}$};
\draw[bvec] (0,-1.8)   -- ++(0.8,0) node[below right] {$\mathbf{B}$};

\end{tikzpicture}
