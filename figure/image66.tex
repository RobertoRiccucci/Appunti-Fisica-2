\begin{tikzpicture}[
  >={Latex[length=3mm,width=2.2mm]},
  every node/.style={font=\small},
  axis/.style={->,thick,black},
  evec/.style={->,line width=1pt,ered},
  bvec/.style={->,line width=1pt,bblue}
]

% Assi
\draw[axis] (0,-2.0) -- (0,2.5) node[above] {$x$};
\draw[axis] (-0.5,0) -- (7.5,0) node[right] {$z$};
\draw[axis] (0,0) -- (-1.2,-1.2) node[below left] {$y$};

% Onda E (rosso, nel piano xz)
\draw[ered,thick,smooth,domain=0:7,samples=80]
  plot (\x,{1.3*sin(120*\x)});
% Frecce E
\foreach \x in {0.0,0.75,1.5,2.25,3.0,3.75,4.5,5.25,6.0,6.75}{
  \draw[evec,thin] (\x,0) -- (\x,{1.3*sin(120*\x)});
}
\node[ered,above right] at (0.5,1.3) {$\mathbf{E}$};
\node[ered,left] at (0,1.3) {$E_0$};

% Onda B (blu, nel piano yz, sfasata di 90 gradi nel rendering)
\foreach \x in {0.0,0.75,1.5,2.25,3.0,3.75,4.5,5.25,6.0,6.75}{
  \pgfmathsetmacro{\bval}{0.8*sin(120*\x)}
  \draw[bvec,thin] (\x,0) -- ({\x+\bval*0.4},{\bval*(-0.4)});
}
% Curva B
\draw[bblue,thick,smooth,domain=0:7,samples=80]
  plot ({\x+0.8*sin(120*\x)*0.4},{0.8*sin(120*\x)*(-0.4)});
\node[bblue,below right] at (0.5,-0.6) {$\mathbf{B}$};
\node[bblue,left] at (-0.5,-0.5) {$E_0/c$};

% Freccia c (direzione di propagazione)
\draw[->,thick] (5.5,1.5) -- (6.5,1.5) node[right] {$c$};

\end{tikzpicture}
